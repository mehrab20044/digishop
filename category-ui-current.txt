function useStore(path) {
  const [attempt, setAttempt] = useState(0);
  const [result, setResult] = useState({ key: '', data: null, error: null });
  const key = `${path}:${attempt}`;
  useEffect(() => {
    const controller = new AbortController();
    let active = true;
    let timedOut = false;
    const timer = setTimeout(() => { timedOut = true; controller.abort(); }, 15000);
    getStore(path, controller.signal)
      .then(data => { if (active) setResult({ key, data, error: null }); })
      .catch(error => {
        if (active) setResult({ key, data: null, error: timedOut ? new Error('پاسخ سرور طول کشید. دوباره تلاش کن.') : error });
      })
      .finally(() => clearTimeout(timer));
    return () => { active = false; clearTimeout(timer); controller.abort(); };
  }, [path, key]);
  return { data: result.key === key ? result.data : null, error: result.key === key ? result.error : null, retry: () => setAttempt(value => value + 1) };
}

function ProductDetail({ id }) {
  const result = useStore(`products/${id}/`);
  if (result.error) return <Failure error={result.error} retry={result.retry} />;
  if (!result.data) return <div className="sf-message" role="status">در حال دریافت محصول…</div>;
  return <ProductContent key={id} product={result.data} />;
}

function ProductContent({ product }) {
  const gallery = (product.images || [])
    .map((image) => ({
      ...image,
      image: safeImage(image.image),
    }))
    .filter((image) => image.image)
    .sort(
      (a, b) =>
        Number(b.is_primary) - Number(a.is_primary) ||
        a.sort_order - b.sort_order ||
        a.id - b.id,
    );

  if (!gallery.length && safeImage(product.image)) {
    gallery.push({
      id: "legacy",
      image: safeImage(product.image),
    });
  }

  const variants = (product.variants || []).filter(
    (variant) => variant.is_active,
  );

  const [selectedId, setSelectedId] = useState(
    product.default_variant?.id || variants[0]?.id || null,
  );

  const [imageIndex, setImageIndex] = useState(0);

  const selected = variants.find(
    (variant) => variant.id === selectedId,
  );

  const price = selected ? selected.price : product.effective_price;

  const available = Boolean(
    product.is_available &&
      (!selected || (selected.stock > 0 && selected.price > 0)),
  );

  return (
    <>
      <nav className="sf-breadcrumb" aria-label="مسیر صفحه">
        <a href="/">فروشگاه</a>

export function useCategoryTree() {
  const [data, setData] = useState(null);
  const [error, setError] = useState("");
  const [attempt, setAttempt] = useState(0);

  useEffect(() => {
    function refresh() {
      if (document.visibilityState !== "hidden") {
        setAttempt((value) => value + 1);
      }
    }

    window.addEventListener("focus", refresh);
    document.addEventListener("visibilitychange", refresh);

    return () => {
      window.removeEventListener("focus", refresh);
      document.removeEventListener("visibilitychange", refresh);
    };
  }, []);

  useEffect(() => {
    const controller = new AbortController();
    let active = true;
    let timeout = false;

    const timer = setTimeout(() => {
      timeout = true;
      controller.abort();
    }, 15000);

    setError("");

    async function load() {
      const response = await fetch(
        `${API_BASE}/categories/tree/`,
        {
          signal: controller.signal,
          cache: "no-store",
          credentials: "omit",
          headers: { Accept: "application/json" },
        },
      );

      const result = await response.json().catch(() => null);

      if (!response.ok) {
        throw new Error(
          result?.error?.message ||
            "دریافت دسته‌بندی‌ها انجام نشد.",
        );
      }

      if (!Array.isArray(result)) {
        throw new Error("ساختار دسته‌بندی‌ها معتبر نیست.");
      }

      if (active) setData(result);
    }

    load()
      .catch((err) => {
        if (active) {
          setError(
            timeout
              ? "دریافت دسته‌بندی‌ها طول کشید؛ دوباره تلاش کنید."
              : err.message,
          );
        }
      })
      .finally(() => clearTimeout(timer));

    return () => {
      active = false;
      clearTimeout(timer);
      controller.abort();
    };
  }, [attempt]);

  return {
    data,
    error,
    retry: () => setAttempt((value) => value + 1),
  };
}

export function flattenCategoryTree(roots) {
  const rows = [];
  const seen = new Set();

  function walk(node, ancestors) {
    if (seen.has(node.id)) return;

    seen.add(node.id);

    const path = [
      ...ancestors,
      { id: node.id, name: node.name, url: node.url },
    ];

    rows.push({ ...node, path });

    for (const child of node.children || []) {
      walk(child, path);
    }
  }

  for (const root of roots || []) {
    walk(root, []);
  }

  return rows;
}

export function categoryLink(search, category) {
  if (!category?.url) {
    return catalogHref(search, {
      category: category?.id || null,
      categories: null,
      page: null,
    });
  }

  const params = new URLSearchParams(search);

  for (const key of ["category", "categories", "page"]) {
    params.delete(key);
  }

  return (
    category.url +
    (params.size ? `?${params}` : "")
  );
}

        if (!canonical) { canonical = document.createElement("link"); canonical.rel = "canonical"; document.head.append(canonical); }
        canonical.href = new URL(data.url, window.location.origin).href;
        const params = new URLSearchParams(window.location.search);
        params.delete("categories"); params.set("category", data.id);
        setResult(params.toString()); setError("");
      }).catch(err => { if (active && err.name !== "AbortError") setError(err.message); });
    return () => { active = false; controller.abort(); };
  }, [attempt]);
  if (error) return <div className="sf-message" role="alert"><p>{error}</p><button onClick={() => setAttempt(value => value + 1)}>تلاش دوباره</button><a href="/products">همهٔ محصولات</a></div>;
  if (result === null) return <div className="sf-message" role="status">در حال دریافت دسته‌بندی…</div>;
  return renderCatalog(result);
}

export const API_BASE = (import.meta.env?.VITE_STORE_API_BASE || '/products/api/v1').replace(/\/$/, '');
export const PRICE_UNIT = import.meta.env?.VITE_STORE_PRICE_UNIT || 'تومان';

export function number(value) {
  return new Intl.NumberFormat('fa-IR').format(value);
}

export function money(value) {
  if (value === null || value === undefined || value === '') return 'قیمت در دسترس نیست';
  const amount = Number(value);
  return Number.isFinite(amount) ? `${number(amount)} ${PRICE_UNIT}` : 'قیمت در دسترس نیست';
}

export function safeImage(value) {
  if (typeof value !== 'string' || !value) return null;
  try {
    const url = new URL(value, window.location.origin);
    return ['http:', 'https:'].includes(url.protocol) ? url.href : null;
  } catch { return null; }
}

export function cover(product) {
  return safeImage(product.images?.find(image => image.is_primary)?.image)
    || safeImage(product.images?.[0]?.image) || safeImage(product.image);
}

export function pageNumber(value) {
  const parsed = Number(value);
  return Number.isSafeInteger(parsed) && parsed > 0 ? parsed : 1;
}

export function catalogHref(search, changes = {}) {
  const params = new URLSearchParams(search);
  for (const [key, value] of Object.entries(changes)) {
    if (value === '' || value === null || value === undefined) params.delete(key);
    else params.set(key, String(value));
  }
  const query = params.toString();
  return `/products${query ? `?${query}` : ''}`;
}

export async function getStore(path, signal) {
  const response = await fetch(`${API_BASE}/${path}`, {
    headers: { Accept: 'application/json' },
    credentials: 'omit',
    signal,
  });
  const json = await response.json().catch(() => null);
  if (!response.ok) {
    const error = new Error(response.status === 404 ? 'محصول موردنظر پیدا نشد.'
      : typeof json?.detail === 'string' ? json.detail
      : json?.error?.message || `دریافت اطلاعات ناموفق بود (${response.status}).`);
    error.status = response.status;
    throw error;
  }
  if (json === null) throw new Error('پاسخ API معتبر نیست. تنظیمات پروکسی محصولات را بررسی کن.');
  return json;
}
